\section{Дипольний момент системи зарядів}\hypertarget{dipole-moment}{}
%% --------------------------------------------------------

Будь-яка матерія складається із заряджених частинок --- електронів і атомних ядер, --- тому кожне тіло певною мірою і створює електричне поле, і
реагує на зовнішнє. Щоб описати цю взаємодію, часто недостатньо знати лише сумарний заряд системи $q = \sum_i q_i$; важливу роль відіграє й те, як
заряди розташовані один відносно одного. На додаток, можливий і крайній випадок: сумарний заряд може дорівнювати нулю ($q=0$), і все ж система
зарядів не буде \enquote{невидимою} для зовнішнього спостерігача --- вона однаково створює поле в просторі й реагує на зовнішнє поле, просто ця
взаємодія визначається вже не сумарним зарядом, а саме взаємним розташуванням зарядів. Найпростішою кількісною мірою такого взаємного розташування зарядів слугує \emphx{дипольний момент}\index{Дипольний момент} --- векторна сума добутків зарядів на їхні
радіус-вектори:
\begin{equation}\label{eq:el_momentum}
	\vect{p} = \sum_{i=1}^{n}q_i\vect{r}_i,
\end{equation}
де $\vect{r}_i$ --- радіус-вектор $i$-го заряду.

Дипольний момент --- перша (після сумарного заряду) характеристика в такому описі; якщо і сумарний заряд, і дипольний момент виявляються нульовими, слід звернутися до наступного члена розкладу --- квадрупольного моменту \eqref{eq:quadrupole_momentum}.


Загалом дипольний момент залежить від вибору початку відліку: якщо перенести початок
координат на вектор $\vect{a}$, то $\vect{r}_i \to \vect{r}_i - \vect{a}$ і
$\vect{p} \to \vect{p} - q\vect{a}$, де $q = \sum_i q_i$. Отже, якщо система в~цілому
електронейтральна ($q = 0$), то дипольний момент $\vect{p}$ визначений однозначно.

Найпростішою й водночас найважливішою системою з ненульовим дипольним моментом є \emphx{диполь}\index{Диполь електричний} --- система, що складається з
двох точкових зарядів $q$, однакових за величиною і протилежних за знаком (рис.~\ref{tikz:el_dipole}).


\emphx{Плече диполя} --- вектор ($\symbfscr{l}$), що йде від \enquote{$-$} заряду до
\enquote{$+$}, довжина якого дорівнює відстані між зарядами.


\begin{SCfigure}[][h!]
	\centering
	\localinput{elmomentum.tikz}
	\caption{Електричний диполь: $\symbfscr{l}$~--- плече, $\vect r_\pm$~--- радіус-вектори
		зарядів $\pm q$}
	\label{tikz:el_dipole}
\end{SCfigure}
\emphx{Дипольним моментом} диполя називається вектор
\begin{equation}
	\vect{p} = q\symbfscr{l}.
\end{equation}


\emphx{Жорсткий диполь}\index{Диполь електричний!жорсткий} --- диполь із фіксованою відстанню між зарядами, що не
змінюється під зовнішніми впливами.


\emphx{Пружний диполь}\index{Диполь електричний!пружний} --- диполь, де відстань між зарядами змінюється під зовнішніми
впливами, і дипольний момент може змінюватися.

Якщо необхідно визначити поле диполя на далеких відстанях, то зручно користуватись концепцією елементарного, або точкового диполя.
\emphx{Елементарним, або точковим диполем}\index{Диполь електричний!точковий} називають граничну систему з $\ell \to 0$, $|q| \to \infty$ за скінченної величини $p$.

\section{Потенціал точкового диполя}\label{sec:DipolePotential}\hypertarget{point-dipole-potential}{}
%% --------------------------------------------------------

Знайдемо поле диполя на далеких відстанях $r$, тобто за умови $\ell \ll r$; у~такому разі
диполь можна вважати точковим. Почнемо з~потенціалу: це скаляр, і~знайти його простіше, ніж
поле. Розташуємо диполь у~початку координат (рис.~\ref{tikz:dipolefield}). Потенціал
у~точці спостереження $P$ за принципом суперпозиції
\begin{equation*}
	\phi = \frac{q}{|\vect r-\vect r_+|} - \frac{q}{|\vect r-\vect r_-|},\qquad
	\vect r_\pm = \pm\frac{\symbfscr{l}}{2},
\end{equation*}
де $\vect r$~--- радіус-вектор точки $P$, а~$\vect r_\pm$~--- радіус-вектори зарядів $\pm q$
(плече $\symbfscr{l}$ напрямлене від $-q$ до $+q$).

\begin{SCfigure}[0.75][h!]
	\centering
	\localinput{dipolefield.tikz}
	\caption{До виведення потенціалу та напруженості поля диполя}
	\label{tikz:dipolefield}
\end{SCfigure}

Оскільки $r_\pm=\ell/2\ll r$, кожен доданок лише трохи відрізняється від $q/r$. Різницю дає формула
для зміни функції при малому зміщенні~\eqref{eq:shift}, застосована до $f(\vect r)=1/r$
зі зміщенням $\vect\epsilon=-\vect a$:
\begin{equation}\label{eq:shift_first_order}
	\frac1{|\vect r-\vect a|} \approx \frac1r - \vect a\cdot\vect\nabla\frac1r
	= \frac1r + \frac{\vect a\cdot\vect r}{r^3},\qquad |\vect a|\ll r,
\end{equation}
бо $\vect\nabla(1/r) = -\vect r/r^3$. Узявши $\vect a=\vect r_\pm=\pm\symbfscr{l}/2$, маємо
\begin{equation*}
	\phi \approx q\left[\left(\frac1r+\frac{\symbfscr{l}\cdot\vect r}{2r^3}\right)-\left(\frac1r-\frac{\symbfscr{l}\cdot\vect r}{2r^3}\right)\right]
	= \frac{q\,\symbfscr{l}\cdot\vect r}{r^3}.
\end{equation*}
Кулонівські доданки $q/r$ двох зарядів скорочуються, тому потенціал диполя спадає швидше за
кулонівський. У~границі точкового диполя ($\ell\to0$, $\vect p=q\symbfscr{l}$ скінченний)
відкинуті члени зникають, і~рівність стає точною:
\begin{equation}\label{eq:potential_dipole}
	\tcbhighmath{
		\phi = \frac{\vect{p}\cdot\vect{r}}{r^3} = \frac{p\cos\theta}{r^2},
	}
\end{equation}
де $\theta$~--- кут між $\vect p$ і~$\vect r$. Отже, потенціал точкового диполя спадає
з~відстанню як $1/r^2$.

%% --------------------------------------------------------
\section{Поле точкового диполя}\label{sec:DipoleField}\hypertarget{point-dipole-field}{}
%% --------------------------------------------------------

Напруженість знайдемо з~потенціалу за формулою $\Efield=-\vect\nabla\phi$. Запишемо
$\phi=(\vect p\cdot\vect r)\,r^{-3}$ і~застосуємо правило диференціювання добутку:
\begin{equation*}
	\vect\nabla\phi=r^{-3}\,\vect\nabla(\vect p\cdot\vect r)+(\vect p\cdot\vect r)\,\vect\nabla r^{-3}.
\end{equation*}
Оскільки $\vect p\cdot\vect r=p_x x+p_y y+p_z z$, де $\vect p$~--- сталий вектор, то частинні
похідні за $x$, $y$, $z$ дорівнюють $p_x$, $p_y$, $p_z$, тобто $\vect\nabla(\vect p\cdot\vect
r)=\vect p$. Далі, $r=\sqrt{x^2+y^2+z^2}$, тому $\partial r/\partial x=x/r$, і~аналогічно для
$y$ та $z$; отже, $\vect\nabla r=\vect r/r$ і
\begin{equation*}
	\vect\nabla r^{-3}=-3r^{-4}\,\vect\nabla r=-\frac{3\vect r}{r^5}.
\end{equation*}
Підставляючи, одержуємо
\begin{equation}\label{eq:dipole_field}
	\tcbhighmath{
		\Efield = \frac{3(\vect{p}\cdot\vect{r})\vect{r}}{r^5} -
		\frac{\vect{p}}{r^3}.
	}
\end{equation}
Напруженість поля точкового диполя спадає з~відстанню як $1/r^3$. Поле має дві складові:
радіальну, пропорційну $\vect p\cdot\vect r$, і~складову вздовж $\vect p$. На осі диполя
($\vect r\parallel\vect p$) поле напрямлене вздовж $\vect p$ і~дорівнює $2p/r^3$, а~на
серединному перпендикулярі ($\vect r\perp\vect p$)~--- напрямлене проти $\vect p$ і~вдвічі
слабше, $p/r^3$ (задача~\ref{prb:DipolePolar}).

Картину поля показано на рис.~\ref{tikz:dipole_lines}. Силові лінії виходять із~диполя
в~бік $\vect p$, огинають його і~повертаються з~протилежного боку; у~площині, що містить
$\vect p$, їх рівняння $r = C\sin^2\theta$ (задача~\ref{prb:DipolePolar}). Еквіпотенціальні
лінії $\phi = p\cos\theta/r^2 = \const$ скрізь перпендикулярні до силових. Площина,
перпендикулярна до $\vect p$, є еквіпотенціальною поверхнею з~$\phi = 0$.

\begin{figure}[tbp]
	\centering
	\localinput{dipole_lines.tikz}
	\caption{Силові лінії (суцільні) та еквіпотенціальні лінії (штрихові) точкового
		диполя $\vect p$ у~площині, що його містить}
	\label{tikz:dipole_lines}
\end{figure}

Дипольні моменти молекул зручно вимірювати в~\emph{дебаях}: $1~\text{Д}=10^{-18}$~СГСЕ$\cdot$см,
тобто заряд електрона, помножений на відстань близько $0{,}2$~\AA. Наприклад, у~молекули
\ce{HCl} $p\approx1{,}1$~Д, у~молекули води $p\approx1{,}8$~Д (табл.~\ref{tab:dipoles}).
Поле такого диполя на відстанях порядку нанометра сягає $10^{7}$--$10^{8}$~В/м
(задача~\ref{prb:DipoleHClEstimate}).

\section{Потенціал системи зарядів на далеких відстанях}\hypertarget{far-field-potential}{}
%% --------------------------------------------------------

Ми\index{Мультипольне розкладання} знайшли поле й потенціал диполя --- системи з двох зарядів. Узагальнимо цей результат на довільну систему зарядів $\{q_i\}$, що займає певну скінченну область у просторі (рис.~\ref{tikz:far_potential}).

%---------------------------------------------------------
\begin{figure}[h!]\centering
	\localinput{far_potential.tikz}
	\caption{До визначення потенціалу на далеких відстанях}
	\label{tikz:far_potential}
\end{figure}
%---------------------------------------------------------

Потенціал поля в точці спостереження $P$ дорівнює сумі потенціалів, створюваних усіма зарядами:
\begin{equation*}
	\phi(\vect{r}) = \sum_i \frac{q_i}{|\vect{r} - \vect{r}_i|}.
\end{equation*}
Виберемо початок системи координат усередині системи зарядів і знайдемо потенціал на відстанях, набагато більших за розмір області, в якій
локалізовані заряди ($r_i \ll r$). У~\S~\ref{sec:DipolePotential} ми зберігали в розкладі $1/|\vect r-\vect r_i|$ лише члени першого порядку малості за $r_i/r$ \eqref{eq:shift_first_order} --- для диполя цього досить. Але якщо $q=0$ і $\vect p=0$, ці члени нічого не дають, тому потрібне наступне наближення. Отримаємо його, розклавши $|\vect r-\vect r_i|^{-1}$ точніше, з~квадратичними доданками:
\begin{equation*}
	\frac1{|\vect r-\vect r_i|} = \frac1r\,(1+\epsilon)^{-1/2}, \qquad
	\epsilon = -\frac{2\,\vect r\cdot\vect r_i}{r^2}+\frac{r_i^2}{r^2}\ll1.
\end{equation*}
Розкладаємо за малим $\epsilon$: $(1+\epsilon)^{-1/2}\approx1-\frac{\epsilon}2+\frac{3\epsilon^2}8$. Доданок $-2\,\vect r\cdot\vect r_i/r^2$ у $\epsilon$ має перший порядок за $r_i/r$, а $r_i^2/r^2$ --- другий, тому в $\epsilon^2$ залишаємо лише $4(\vect r\cdot\vect r_i)^2/r^4$:
\begin{equation*}
	(1+\epsilon)^{-1/2}\approx 1+\frac{\vect r\cdot\vect r_i}{r^2}-\frac{r_i^2}{2r^2}+\frac{3(\vect r\cdot\vect r_i)^2}{2r^4}.
\end{equation*}
Доданок $-r_i^2/(2r^2)$ породжує $-r^2r_i^2$ у чисельнику третього члена; саме він у підсумку робить тензор квадрупольного моменту безслідовим (див. нижче). Помноживши на $1/r$, маємо розклад до другого порядку за $r_i/r$ включно:
\begin{equation}\label{eq:multipole_second_order}
	\frac1{|\vect r - \vect r_i|} \approx \frac1r + \frac{\vect r\cdot\vect r_i}{r^3} +
	\frac{3(\vect r\cdot\vect r_i)^2 - r^2 r_i^2}{2r^5}.
\end{equation}
Підставляючи \eqref{eq:multipole_second_order} у $\phi(\vect r) = \sum_i q_i/|\vect r - \vect r_i|$, одержуємо
\begin{equation*}
	\phi(\vect{r}) \approx \frac1r\sum_i q_i + \frac{\vect r}{r^3}\cdot\sum_i q_i\vect r_i +
	\frac1{2r^5}\sum_i q_i\left[3(\vect r\cdot\vect r_i)^2 - r^2 r_i^2\right].
\end{equation*}

Перші дві суми --- це вже знайомі нам повний заряд і дипольний момент системи:
\begin{equation*}
	q = \sum_i {q_i}, \quad \vect{p} = \sum_i q_i\vect{r}_i.
\end{equation*}
Третю суму розпишемо покомпонентно, позначаючи декартові компоненти індексами $\alpha,\beta = x,y,z$ (за індексами, що повторюються, мається на
увазі підсумовування):
\begin{equation*}
	(\vect r\cdot\vect r_i)^2 = r_\alpha r_\beta\, r_{i\alpha} r_{i\beta}, \qquad
	r^2 r_i^2 = r_\alpha r_\beta\, \delta_{\alpha\beta}\, r_i^2,
\end{equation*}
де $\delta_{\alpha\beta}$ --- символ Кронекера. Тоді
\begin{equation*}
	\frac1{2r^5}\sum_i q_i\left[3(\vect r\cdot\vect r_i)^2 - r^2 r_i^2\right] =
	\frac1{2r^5}\, r_\alpha r_\beta \sum_i q_i\left(3r_{i\alpha}r_{i\beta} - r_i^2\delta_{\alpha\beta}\right).
\end{equation*}
Суму за $i$ в~правій частині називають \emphx{тензором квадрупольного моменту}\index{Квадрупольний момент} системи зарядів:
\begin{equation}\label{eq:quadrupole_momentum}
	\tcbhighmath{
		D_{\alpha\beta} = \sum_i q_i\left(3r_{i\alpha}r_{i\beta} - r_i^2\delta_{\alpha\beta}\right).
	}
\end{equation}
Це симетричний тензор другого рангу ($D_{\alpha\beta}=D_{\beta\alpha}$), причому сума його діагональних елементів (слід) дорівнює нулю:
\begin{equation*}
	\sum_\alpha D_{\alpha\alpha} = \sum_i q_i\left(3r_i^2 - 3r_i^2\right) = 0,
\end{equation*}
тому серед дев'яти компонент $D_{\alpha\beta}$ лише п'ять незалежних. Якщо перейти у власну систему координат тензора (в якій він діагональний),
то залишаються дві незалежні діагональні компоненти; решту інформації несуть три кути, що задають орієнтацію власних осей.

Отже, повний розклад потенціалу системи зарядів на далеких відстанях набуває вигляду
\begin{equation}\label{eq:multipoles}
	\phi(\vect r) = \underbrace{\frac{q}r}_{\text{монополь}} +
	\underbrace{\frac{\vect p\cdot\vect r}{r^3}}_{\text{диполь}} +
	\underbrace{\frac1{2r^5}\sum_{\alpha\beta} D_{\alpha\beta}\, r_\alpha r_\beta}_{\text{квадруполь}} + \ldots
\end{equation}

Проаналізуємо цей розклад. Кожен наступний доданок спадає з відстанню швидше за попередній: $\phi\sim1/r$ для монополя, $\phi\sim1/r^2$ для диполя і
$\phi\sim1/r^3$ для квадруполя (відповідно й напруженість поля $\Efield=-\vect\nabla\phi\sim1/r^4$ спадає швидше, ніж дипольна
$\sim1/r^3$). Тому внесок у поле на далеких відстанях фактично дає лише перший доданок розкладу, що не занулюється: при $q\neq0$ домінує монополь; при
$q=0,\ \vect p\neq0$ --- диполь; при $q=0$ і $\vect p=0$ --- квадруполь. Саме тому систему
зарядів, у~якої $q=0$, $\vect p = 0$, а~$D_{\alpha\beta}\neq0$, називають \emphx{квадруполем}.

Молекули здебільшого електронейтральні, а деякі з них (наприклад, \ce{H2O}, \ce{HCl}) мають ненульовий дипольний момент. Інші,
такі як лінійна симетрична молекула \ce{CO2}, дипольного моменту не мають ($\vect p = 0$) через симетрію розташування атомів кисню відносно атома
вуглецю, проте мають ненульовий квадрупольний момент. Тому далекодійна електростатична взаємодія таких молекул між собою і з іншими зарядами визначається
саме квадруполь-квадрупольною або квадруполь-дипольною взаємодією і є значно слабшою, ніж диполь-дипольна.

\begin{SCfigure}[2][tbp]
	\centering
	\localinput{linear_quadrupole.tikz}
	\caption{Лінійний квадруполь: заряди $+q$, $-2q$, $+q$ на осі $z$}
	\label{tikz:linear_quadrupole}
\end{SCfigure}

\begin{Example}{Лінійний квадруполь}
	Розглянемо три заряди, розташовані на осі $Oz$ (рис.~\ref{tikz:linear_quadrupole}): заряди $+q$ у точках $z=\pm a$ та заряд $-2q$ у початку координат $z=0$ --- грубу модель лінійної
	неполярної молекули на кшталт \ce{CO2}. (У~реальній молекулі \ce{CO2} знаки протилежні: атоми кисню заряджені від'ємно, а атом
	вуглецю~--- додатно; це змінює лише знак $D_{\alpha\beta}$.)

	Сумарний заряд системи дорівнює нулю:
	\begin{equation*}
		q_\Sigma = q + (-2q) + q = 0.
	\end{equation*}
	Дипольний момент також дорівнює нулю через симетрію розташування зарядів відносно початку координат:
	\begin{equation*}
		\vect p = q\,a\vect e_z + (-2q)\cdot 0 + q\,(-a)\vect e_z = 0.
	\end{equation*}
	Отже, поле на далеких відстанях визначається квадрупольним моментом. Обчислимо компоненти тензора $D_{\alpha\beta}$. Оскільки всі заряди лежать на осі
	$z$, для кожного заряду $r_{ix}=r_{iy}=0$, $r_{iz}=z_i$, $r_i^2=z_i^2$. Тому всі недіагональні компоненти дорівнюють нулю, а
	\begin{equation*}
		D_{zz} = \sum_i q_i\left(3z_i^2 - z_i^2\right) = 2\sum_i q_i z_i^2
		= 2\left[q\,a^2 + (-2q)\cdot 0 + q\,a^2\right] = 4qa^2,
	\end{equation*}
	\begin{multline*}
		D_{xx} = \sum_i q_i\left(3\cdot0 - z_i^2\right) = -\sum_i q_i z_i^2 = -\left(qa^2 + qa^2\right) = -2qa^2, \\ D_{yy}=D_{xx}=-2qa^2
	\end{multline*}
	(перевірка сліду: $D_{xx}+D_{yy}+D_{zz} = -2qa^2-2qa^2+4qa^2=0$ --- узгоджується із загальною властивістю). Оскільки тензор уже діагональний, для точки
	спостереження на осі $z$ (де $\vect r = z\vect e_z$) залишається лише доданок з $D_{zz}$:
	\begin{equation*}
		\phi_{\text{кв}}(z) = \frac{D_{zz}}{2z^3} = \frac{2qa^2}{z^3}.
	\end{equation*}
\end{Example}


%% --------------------------------------------------------
\section{Диполь в електричному полі}\label{sec:DipoleEnergy}\hypertarget{dipole-in-field}{}
%% --------------------------------------------------------

Досі\index{Диполь електричний!у зовнішньому полі} ми цікавились полем, яке диполь \emph{створює}. Тепер розглянемо обернену задачу: як диполь поводиться, потрапивши у \emph{зовнішнє} електричне поле --- яка сила й момент сил на нього діють і яку енергію він при цьому має.


%% --------------------------------------------------------
\subsection*{Момент сил, що діє на диполь в однорідному полі}
%% --------------------------------------------------------


Розглянемо диполь в однорідному електричному полі (рис.~\ref{tikz:Momentum_on_dipole_in_yniform_field}). Результуюча сила, що діє на цей диполь, дорівнює
нулю: $\vect{F} = (+q)\Efield + (-q)\Efield = 0$.

\begin{SCfigure}[0.75][h!]
	\centering
	\localinput{DipoleInUF.tikz}
	\caption{До виведення моменту сил, що діє на диполь в однорідному електричному полі}
	\label{tikz:Momentum_on_dipole_in_yniform_field}
\end{SCfigure}

Однак цей диполь може орієнтуватися у такому полі, оскільки на нього може діяти обертовий момент сил. Розрахуємо його відносно довільної точки $O$,
використовуючи означення, відоме з~курсу механіки ($\vect r_\pm$~--- радіус-вектори зарядів
$\pm q$, проведені з~точки $O$):
\begin{equation*}
	\vect{M}_O = \vect{r}_+ \times \vect{F}_+ +   \vect{r}_- \times \vect{F}_- = \vect{r}_+ \times (+q)\Efield +   \vect{r}_- \times (-q)\Efield =
	q(\vect{r}_+ - \vect{r}_-)\times \Efield.
\end{equation*}
Оскільки $q(\vect{r}_+ - \vect{r}_-) = \vect{p}$, то остаточно маємо:


\begin{equation*}
	\tcbhighmath{
		\vect{M} = \vect{p}\times\Efield.
	}
\end{equation*}


Бачимо, що момент сил (пари сил) не залежить від вибору точки $O$: так і має бути, адже
результуюча сила дорівнює нулю. За модулем $M = pE\sin\theta$, де $\theta$~--- кут між
$\vect{p}$ і~$\Efield$. Цей момент повертає диполь так, щоб $\vect{p}$ став паралельним
полю.


%% --------------------------------------------------------
\subsection*{Енергія жорсткого диполя в електричному полі}
%% --------------------------------------------------------

Енергія точкового заряду $q$ у~зовнішньому полі дорівнює $U = q\phi$, де $\phi$~---
потенціал поля в~точці, де перебуває заряд $q$. Диполь~--- це система з~двох зарядів, тому
його енергія в~зовнішньому полі $U= q\phi_+ + (-q)\phi_- = q (\phi_+ - \phi_-)$, де
$\phi_\pm$~--- потенціал у~точках, де розташовані заряди $\pm q$. Ці точки зміщені одна
відносно одної на плече $\symbfscr{l}$, тож за формулою~\eqref{eq:shift}
\begin{equation*}
	U = q (\phi_+ - \phi_-) = q\, \symbfscr{l}\cdot\vect{\nabla}\phi.
\end{equation*}
Оскільки $-\vect{\nabla}\phi = \Efield$, а~$\vect{p} = q\symbfscr{l}$:
\begin{equation}\label{eq:DipoleEnergy}
	\tcbhighmath{
		U = -\vect{p}\cdot\Efield.
	}
\end{equation}

Енергія $U = -pE\cos\theta$ мінімальна в~положенні $\vect{p} \uparrow\uparrow \Efield$
(положення стійкої рівноваги). У~разі відхилення з~цього положення виникає момент сил, що
повертає диполь до положення рівноваги (рис.~\ref{tikz:angle}). Положення $\vect{p}
\uparrow\downarrow \Efield$ теж є положенням рівноваги ($\vect M = 0$), але нестійкої: енергія
в~ньому максимальна, і~найменше відхилення призводить до повороту диполя.
\begin{figure}[h!]
	\centering
	\begin{subcaptionblock}{.45\textwidth}
		\centering
		\localinput{parallel.tikz}
		\caption{}
		\label{tikz:parallel}
	\end{subcaptionblock}
	\begin{subcaptionblock}{.45\textwidth}
		\centering
		\localinput{angle.tikz}
		\caption{}
		\label{tikz:angle}
	\end{subcaptionblock}
	\caption{Диполь в однорідному електричному полі. Ліворуч (\subref{tikz:parallel}) --- положення стійкої рівноваги з мінімальною енергією.
		Праворуч (\subref{tikz:angle}) --- диполь під кутом до поля: момент сил повертає його до положення~(\subref{tikz:parallel}).}
\end{figure}



%% --------------------------------------------------------
\subsection*{Сила, що діє на диполь у неоднорідному полі}
%% --------------------------------------------------------

В~однорідному полі енергія~\eqref{eq:DipoleEnergy} не залежить від положення диполя, і~сили
немає. У~неоднорідному ж полі (рис.~\ref{tikz:dipole_in_nonuniform_field}) поля в~місцях
зарядів $\pm q$ різні:
\begin{equation*}
	\vect{F} = (+q)\Efield_+ + (-q)\Efield_- = q(\Efield_+ - \Efield_-).
\end{equation*}

\begin{SCfigure}[1.5][h!]
	\centering
	\localinput{dipinnonuniformfield.tikz}
	\caption{Диполь у неоднорідному полі}
	\label{tikz:dipole_in_nonuniform_field}
\end{SCfigure}

Застосувавши формулу~\eqref{eq:shift} до кожної компоненти поля зі зміщенням $\symbfscr{l}$,
маємо $\Efield_+ - \Efield_- = (\symbfscr{l}\cdot\vect\nabla)\Efield$, і
\begin{equation}\label{eq:EforceOnDipole}
	\tcbhighmath{
		\vect{F} = p_x \frac{\partial\Efield}{\partial x} +
		p_y \frac{\partial\Efield}{\partial y} + p_z \frac{\partial\Efield}{\partial z}
		=
		(\vect{p}\cdot\vect{\nabla})\Efield.
	}
\end{equation}
Для жорсткого диполя ($\vect{p} = \const$) в~електростатичному полі ($\rot\Efield = 0$) цю
силу можна отримати й~з~енергії: справджується тотожність $(\vect{p}\cdot\vect{\nabla})\Efield
= \vect{\nabla}(\vect{p}\cdot\Efield)$, тому $\vect{F} = -\vect{\nabla}U$, де $U$~--- енергія
диполя~\eqref{eq:DipoleEnergy}. Диполь, орієнтований уздовж поля (саме таку орієнтацію він
набуває під дією моменту $\vect{M}$), має $U = -pE$, і~сила $\vect{F} = p\,\vect{\nabla}E$
\emph{втягує його в~область сильнішого поля}~--- туди, де його енергія нижча. Диполь,
орієнтований проти поля, навпаки, виштовхується з~неї. Саме тому дрібні шматочки паперу,
поляризовані полем натертого бурштину, притягуються до нього.

%% --------------------------------------------------------
\subsection*{Енергія пружного диполя в електричному полі}
%% --------------------------------------------------------

У пружному диполі (моделі атома чи неполярної молекули) за відсутності поля заряди
збігаються, і~дипольного моменту немає. Під дією електричного поля заряди зміщуються один
відносно одного на таку відстань $\ell$, що сила з боку поля врівноважується силою пружності
$
	k\ell = qE
$.
Звідси знаходимо, що дипольний момент $p = q\ell$ залежить від прикладеного поля за законом:
\begin{equation*}
	p = \frac{q^2E}{k} = \alpha E.
\end{equation*}
Коефіцієнт $\alpha$ називається \emphx{поляризовністю молекули}\index{Поляризовність молекули}.


На відміну від жорсткого диполя, тут недостатньо врахувати лише енергію пружної деформації --- потрібна ще й енергія взаємодії заряду з полем. Повна
енергія системи як функція зміщення $x$ складається з цих двох внесків:
\begin{equation*}
	U(x) = \frac{kx^2}2 - qxE.
\end{equation*}
Мінімізуючи за $x$ ($dU/dx = kx - qE = 0$), знаходимо те саме рівноважне зміщення $x=\ell=qE/k$, що й з умови рівноваги сил. Підставляючи його назад,
одержуємо повну енергію в рівновазі:
\begin{equation*}
	U = \frac{k\ell^2}2 - qE\ell = \frac{qE\ell}2 - qE\ell = -\frac{qE\ell}2 = -\frac{pE}2.
\end{equation*}
Тут вектор дипольного моменту, індукований зовнішнім полем, паралельний електричному полю
$\vect{p}\uparrow\uparrow \Efield$. Тому останню формулу можна переписати у векторній формі:

\begin{equation}
	\tcbhighmath{
		U = -\frac12\ \vect{p}\cdot\Efield.
	}
\end{equation}

Знак \enquote{мінус} тут той самий, що й для жорсткого диполя (вирівнювання вздовж поля знижує енергію), але вдвічі менший за модулем: половину
\enquote{виграшу} від взаємодії з полем \enquote{з'їдає} енергія пружної деформації, необхідної для створення самого диполя.


%% --------------------------------------------------------
\section{Енергія довільної системи зарядів у зовнішньому полі*}\label{sec:QuadEnergy}\hypertarget{energy-in-external-field}{}
%% --------------------------------------------------------
\begin{Extra}

Розглянемо систему зарядів $\{q_i\}$, локалізовану в деякій обмеженій області простору, яку внесено у \emph{зовнішнє} електричне поле з потенціалом
$\phi(\vect r)$ і напруженістю $\Efield(\vect r) = -\vect\nabla\phi(\vect r)$. Слово \enquote{зовнішнє} тут ключове: це поле створене \emph{іншими} зарядами,
розташованими \emph{поза} системою $\{q_i\}$ --- система $\{q_i\}$ лише перебуває в цьому вже заданому полі, не будучи його джерелом. Знайдемо
енергію взаємодії системи $\{q_i\}$ із цим зовнішнім полем.

Результат $U=-\vect p\cdot\Efield$, отриманий вище для жорсткого диполя (двох зарядів) у першому порядку за $\ell$, виявиться першим наближенням
загальнішого розкладу, застосовного до довільної системи $\{q_i\}$ у довільному (не обов'язково однорідному) зовнішньому полі. Енергія системи
дорівнює сумі енергій окремих зарядів у цьому полі:
\begin{equation*}
	U = \sum_i q_i\, \phi(\vect r_i).
\end{equation*}
Виберемо початок координат $O$ десь усередині системи й розкладемо $\phi(\vect r_i)$ у ряд Тейлора за $\vect r_i$, аналогічно до того, як це робилося
для потенціалу самої системи (пор.~\eqref{eq:multipole_second_order}):
\begin{equation*}
	\phi(\vect r_i) \approx \phi(0) + \vect r_i\cdot\vect\nabla\phi(0) +
	\frac12\, r_{i\alpha}r_{i\beta}\, \frac{\partial^2\phi}{\partial x_\alpha\partial x_\beta}(0).
\end{equation*}
Підсумовуючи із вагами $q_i$ й використовуючи $\vect\nabla\phi = -\Efield$:
\begin{equation*}
	U \approx \phi(0)\underbrace{\sum_i q_i}_{q} - \Efield(0)\cdot\underbrace{\sum_i q_i\vect r_i}_{\vect p} +
	\frac12\sum_{\alpha\beta}\left[\sum_i q_i\, r_{i\alpha}r_{i\beta}\right]\frac{\partial^2\phi}{\partial x_\alpha\partial x_\beta}(0).
\end{equation*}
Останній доданок хочеться виразити через уже введений тензор $D_{\alpha\beta}$, а не через $\sum_i q_i r_{i\alpha}r_{i\beta}$ напряму. Оскільки в точці
$O$ немає зарядів, що створюють зовнішнє поле (його потенціал там задовольняє рівняння Лапласа), маємо
$
	\nabla^2\phi(0) = \sum_\alpha \dfrac{\partial^2\phi}{\partial x_\alpha^2}(0) = 0,
$
тобто тензор других похідних $\partial^2\phi/\partial x_\alpha\partial x_\beta$ безслідовий --- так само як і $D_{\alpha\beta}$. Це дає змогу
показати (додавши й віднявши слідову частину), що

\begin{multline*}
	\frac12\sum_{\alpha\beta}\left[\sum_i q_i\, r_{i\alpha}r_{i\beta}\right]\frac{\partial^2\phi}{\partial x_\alpha\partial x_\beta}(0)
	= \frac16\sum_{\alpha\beta} D_{\alpha\beta}\, \frac{\partial^2\phi}{\partial x_\alpha\partial x_\beta}(0) = \\
	= -\frac16\sum_{\alpha\beta} D_{\alpha\beta}\, \frac{\partial E_\alpha}{\partial x_\beta}(0),
\end{multline*}
адже доданок, пропорційний $\delta_{\alpha\beta}$, містить $\nabla^2\phi(0) = 0$ і~зникає. Остаточно енергія
довільної системи зарядів у зовнішньому полі --- це ряд за мультипольними моментами системи:
\begin{equation}
	U = \underbrace{q\,\phi(0)}_{\text{монополь}} \;-\; \underbrace{\vect p\cdot\Efield(0)}_{\text{диполь}} \;-\;
	\underbrace{\frac16\sum_{\alpha\beta} D_{\alpha\beta}\, \frac{\partial E_\alpha}{\partial x_\beta}(0)}_{\text{квадруполь}} \;+\;\ldots
\end{equation}

Умова $\nabla^2\phi(0)=0$ тут не тотожність, а~суттєве припущення: вона виконується лише
тому, що $\phi$~--- потенціал \emph{зовнішнього} поля, створеного зарядами поза системою
$\{q_i\}$. Якби $\phi$ включало власне поле системи, у~точці $O$ були б заряди,
$\nabla^2\phi(0) = -4\pi\rho(0) \neq 0$, і~слідовий доданок не скорочувався б.

Кожен доданок відповідає за взаємодію відповідного мультипольного моменту системи з полем
(або його похідними) у точці $O$: заряд взаємодіє з потенціалом, диполь --- із самим
полем, квадруполь --- з градієнтом поля (неоднорідністю), і так далі. Для диполя
$q=0$. Його квадрупольний момент відносно середини відрізка, що з'єднує заряди, дорівнює
нулю, а вищі моменти (октупольний тощо) пропорційні $q\ell^3$ і~вищим степеням $\ell$
й~зникають у~границі точкового диполя ($\ell \to 0$, $p = q\ell$ фіксований). Тому для
точкового диполя в~розкладі залишається лише $U=-\vect p\cdot\Efield$~--- результат,
отриманий вище елементарним способом.
\end{Extra}

\section*{Підсумки}

\addcontentsline{toc}{section}{Підсумки}
%% --------------------------------------------------------

Щоб описати взаємодію системи зарядів з полем, недостатньо знати лише її сумарний заряд: при $q=0$ усе визначає взаємне розташування зарядів. Першою (після заряду) його характеристикою є \emphx{дипольний момент} $\vect p=\sum_i q_i\vect r_i$~\eqref{eq:el_momentum}, який для електронейтральної системи не залежить від вибору початку відліку. Найпростіша система з ненульовим $\vect p$ --- \emphx{диполь}, тобто пара рівних за величиною й протилежних за знаком зарядів, $\vect p=q\symbfscr{l}$. У границі точкового диполя його поле $\Efield=\dfrac{3(\vect p\cdot\vect r)\vect r}{r^5}-\dfrac{\vect p}{r^3}$~\eqref{eq:dipole_field} спадає як $1/r^3$, а потенціал $\phi=\dfrac{\vect p\cdot\vect r}{r^3}$~\eqref{eq:potential_dipole} --- як $1/r^2$, тобто швидше, ніж кулонівські.

Узагальнення цього результату~--- розклад потенціалу довільної системи зарядів на великих відстанях за мультипольними моментами~\eqref{eq:multipoles}: \emphx{монополь} $q/r$, \emphx{диполь} і~\emphx{квадруполь}, який задається симетричним безслідовим тензором $D_{\alpha\beta}$~\eqref{eq:quadrupole_momentum} з~п'ятьма незалежними компонентами. Кожен наступний доданок спадає швидше за попередній, тому на далеких відстанях поле визначає перший з~ненульових моментів. Так, у~молекули \ce{CO2} немає дипольного моменту, і~її електростатична взаємодія визначається квадрупольним моментом.

Ті самі моменти визначають і поведінку системи в \emph{зовнішньому} полі. В однорідному полі на диполь діє нульова сила, але ненульовий момент $\vect M=\vect p\times\Efield$, що орієнтує його вздовж поля. У неоднорідному полі виникає ще й сила $\vect F=(\vect p\cdot\vect\nabla)\Efield$~\eqref{eq:EforceOnDipole}, що втягує диполь в область сильнішого поля. Енергія жорсткого диполя дорівнює $U=-\vect p\cdot\Efield$, а пружного, у якого $\vect p=\alpha\Efield$ ($\alpha$ --- поляризовність), --- $U=-\frac12\vect p\cdot\Efield$: половину виграшу поглинає енергія деформації. Для довільної системи енергія у зовнішньому полі також є рядом за мультипольними моментами: заряд взаємодіє з потенціалом, диполь --- з полем, квадруполь --- з градієнтом поля.

% задачі цього розділу короткі: не розриваємо їх між сторінками

\begin{problem}\label{prb:DipolePolar}
	\emph{Полярні компоненти.}
	Точковий диполь $\vect p$ напрямлений уздовж осі $Oz$; $\theta$ --- кут між $\vect r$ і $\vect p$. Виходячи з $\phi = p\cos\theta/r^2$, знайдіть компоненти $E_r$ і $E_\theta$ поля диполя. Який вигляд має поле на осі диполя ($\theta=0$) і на його серединному перпендикулярі ($\theta=\pi/2$)? Покажіть, що силові лінії мають вигляд $r = C\sin^2\theta$ (рис.~\ref{tikz:dipole_lines}).
	\begin{solution}
		$E_r=\dfrac{2p\cos\theta}{r^3}$, $E_\theta=\dfrac{p\sin\theta}{r^3}$; на осі $E=2p/r^3$ уздовж $\vect p$, на серединному перпендикулярі $E=p/r^3$ проти $\vect p$. Уздовж силової лінії $\dfrac{dr}{r\,d\theta}=\dfrac{E_r}{E_\theta}=\dfrac{2\cos\theta}{\sin\theta}$, звідки $\ln r = 2\ln\sin\theta + \const$, тобто $r = C\sin^2\theta$.
	\end{solution}
\end{problem}

\begin{problem}\label{prb:DipoleMagicAngle}
	\emph{Особливий кут.} 
	Під яким кутом $\theta$ до дипольного моменту $\vect p$ вектор напруженості поля диполя перпендикулярний до $\vect p$?
	\begin{solution}
		$\cos^2\theta=\dfrac13$, тобто $\theta\approx54{,}7^\circ$ (і $125{,}3^\circ$). Підказка: $\vect p\cdot\Efield=\dfrac{p^2}{r^3}\,(3\cos^2\theta-1)$.
	\end{solution}
\end{problem}

\begin{problem}\label{prb:DipoleHClEstimate}
	\emph{Оцінка.}
	Дипольний момент молекули \ce{HCl} приблизно дорівнює $1\,\text{Д}=10^{-18}$~СГСЕ$\cdot$см. Оцініть напруженість поля на осі такого диполя на відстані $r=1$~нм. (Для переходу до СІ: $1$~СГСЕ$_E\approx3\cdot10^{4}$~В/м.)
	\begin{solution}
		$E=2p/r^3=2\cdot10^{-18}/(10^{-7})^3=2\cdot10^{3}$~СГСЕ $\approx6\cdot10^{7}$~В/м~--- у~сотні разів більше, ніж поле пробою повітря.
	\end{solution}
\end{problem}

\begin{problem}\label{prb:SquareQuadrupole}
	\emph{Плоский квадруполь.}
	У площині $Oxy$ у вершинах квадрата зі стороною $a$ розташовано заряди: $+q$ у точці $(0,0)$, $-q$ у $(a,0)$, $+q$ у $(a,a)$, $-q$ у $(0,a)$. Знайдіть сумарний заряд, дипольний момент і компоненти тензора $D_{\alpha\beta}$.
	\begin{solution}
		$q_\Sigma=0$, $\vect p=0$; єдині ненульові компоненти $D_{xy}=D_{yx}=3qa^2$ (слід нульовий).
	\end{solution}
\end{problem}

\begin{problem}\label{prb:QuadrupoleAccuracy}
	\emph{Точність наближення.}
	Для лінійного квадруполя з прикладу ($+q,\,-2q,\,+q$ на осі $Oz$ у точках $z=a,\,0,\,-a$): (а)~знайдіть $E_z(z)$ на осі в квадрупольному наближенні; (б)~обчисліть точне значення потенціалу в точці $z=5a$ і порівняйте його з $\phi_{\text{кв}}=2qa^2/z^3$. Яка відносна похибка?
	\begin{solution}
		(а)~$E_z=6qa^2/z^4$. (б)~Точно $\phi=\dfrac{2qa^2}{z(z^2-a^2)}=\dfrac{q}{60a}$, наближення дає $\dfrac{2q}{125a}$; їхнє відношення $(z^2-a^2)/z^2=24/25$, тож похибка $\approx4\%$ (наближення занижує).
	\end{solution}
\end{problem}

\begin{problem}\label{prb:DipoleOscillations}
	\emph{Коливання диполя.}
	Жорсткий диполь з моментом $p$ і моментом інерції $I$ (відносно осі, перпендикулярної до $\vect p$) може вільно обертатися в однорідному полі $\Efield$. Знайдіть період малих коливань біля положення стійкої рівноваги.
	\begin{solution}
		$T=2\pi\sqrt{I/(pE)}$.
	\end{solution}
\end{problem}

\begin{problem}\label{prb:DipoleRotationWork}
	\emph{Робота повороту.}
	Яку роботу треба виконати, щоб повернути жорсткий диполь в однорідному полі з положення $\vect p\uparrow\uparrow\Efield$ у положення $\vect p\uparrow\downarrow\Efield$? А в положення $\vect p\perp\Efield$?
	\begin{solution}
		$A=2pE$ і $A=pE$ відповідно.
	\end{solution}
\end{problem}

\begin{problem}\label{prb:DipoleAndCharge}
	\emph{Диполь і точковий заряд.}
	Точковий заряд $Q$ створює поле $\Efield=Q\vect r/r^3$. На відстані $r$ від нього розташований жорсткий диполь $\vect p$, напрямлений уздовж радіуса від заряду. Знайдіть силу, що діє на диполь, і перевірте відповідь за допомогою третього закону Ньютона й поля диполя.
	\begin{solution}
		$F_r=p\,\dfrac{d}{dr}\dfrac{Q}{r^2}=-\dfrac{2pQ}{r^3}$, тобто притягання із силою $2pQ/r^3$. Перевірка: поле диполя на осі в місці заряду дорівнює $2p/r^3$, тож сила на заряд $2pQ/r^3$ і вона протилежна.
	\end{solution}
\end{problem}

\begin{problem}\label{prb:TwoDipoles}
	\emph{Взаємодія двох диполів.}
	Два однакових жорстких диполі $p$ з паралельними моментами розташовані на відстані $r$ один від одного. Знайдіть енергію їхньої взаємодії, якщо (а)~$\vect r\parallel\vect p$ (на одній прямій); (б)~$\vect r\perp\vect p$ (поруч). Яке розташування вигідніше?
	\begin{solution}
		(а)~$U=-2p^2/r^3$; (б)~$U=+p^2/r^3$. Вигідніше розташування на одній прямій.
	\end{solution}
\end{problem}

\begin{problem}\label{prb:Polarizability}
	\emph{Поляризовність.}
	Молекулу змодельовано зарядом $e$ і масою $m$ на пружині з власною круговою частотою $\omega_0=10^{16}\ \text{с}^{-1}$. Виразіть поляризовність $\alpha$ через $e$, $m$, $\omega_0$ та оцініть її в $\text{см}^3$.
	\begin{solution}
		$\alpha=\dfrac{e^2}{m\omega_0^2}\approx\dfrac{(4{,}8\cdot10^{-10})^2}{9{,}1\cdot10^{-28}\cdot10^{32}}\approx2{,}5\cdot10^{-24}\ \text{см}^3$ (порядок розміру атома в кубі).
	\end{solution}
\end{problem}

\begin{problem}\label{prb:QuadrupoleInField}
	\emph{Квадруполь у~неоднорідному полі*.}
	Лінійний квадруполь ($+q,\,-2q,\,+q$ у~точках $z=a,\,0,\,-a$) розташовано в~зовнішньому
	полі $\Efield=g\left(-\dfrac x2,\ -\dfrac y2,\ z\right)$, $g=\const$. (а)~Переконайтеся, що
	це поле потенціальне й~вільне від зарядів, і~знайдіть його потенціал, поклавши
	$\phi(0)=0$. (б)~Знайдіть енергію квадруполя безпосередньо, $U=\sum_i q_i\phi(\vect r_i)$.
	(в)~Перевірте відповідь за мультипольною формулою з~\S~\ref{sec:QuadEnergy}.
	\begin{solution}
		(а)~$\rot\Efield=0$, $\divg\Efield=g\left(-\tfrac12-\tfrac12+1\right)=0$; $\phi=-g\left(\dfrac{z^2}2-\dfrac{x^2+y^2}4\right)$. (б)~$U=q\phi(a)+q\phi(-a)-2q\phi(0)=-gqa^2$. (в)~$-\dfrac16\left(D_{xx}\dfrac{\partial E_x}{\partial x}+D_{yy}\dfrac{\partial E_y}{\partial y}+D_{zz}\dfrac{\partial E_z}{\partial z}\right)=-\dfrac16\left(qa^2g+qa^2g+4qa^2g\right)=-gqa^2$.
	\end{solution}
\end{problem}
