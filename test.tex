\begin{tikzpicture}[>=Latex, scale=1.5, line join=round]

    % === ГЕОМЕТРИЧНІ ПАРАМЕТРИ ===
    \def\ang{25}     % кут перспективи (напрямок глибини W)
    \def\Lp{4.0}     % довжина пластин L (незмінна)
    \def\W{1.7}      % глибина (ширина) пластин
    \def\Z{1.0}      % відстань між пластинами (= товщина діелектрика)
    \def\Dx{1.15}    % наскільки діелектрик вже висунуто назовні

    \pgfmathsetmacro{\xIns}{\Lp - \Dx}   % довжина діелектрика, що ще всередині (підпис x)
    \pgfmathsetmacro{\dxh}{\W*cos(\ang)} % горизонтальна проєкція глибини
    \pgfmathsetmacro{\dyh}{\W*sin(\ang)} % вертикальна проєкція глибини

    % ------------------------------------------------------------
    % 1. ПЕРЕДНЯ ГРАНЬ ДІЕЛЕКТРИКА (суцільний сірий).
    %    Зліва (x<\Dx) діелектрика вже немає -- там просто порожнеча,
    %    тому нічого не малюємо (зазор між пластинами лишається білим).
    % ------------------------------------------------------------
    \draw[fill=gray!35] (\Dx,-\Z) rectangle (\Lp+\Dx,0);

    % ------------------------------------------------------------
    % 2. ВИСУНУТА ЧАСТИНА ДІЕЛЕКТРИКА У 3D (верхня та бічна грані)
    % ------------------------------------------------------------
    \draw[fill=gray!35]
        (\Lp,0) -- ++(\ang:\W) -- ++(\Dx,0) -- ++(\ang:-\W) -- cycle;               % верхня грань
    \draw[fill=gray!55]
        (\Lp+\Dx,0) -- ++(\ang:\W) -- ++(0,-\Z) -- ++(\ang:-\W) -- cycle;           % бічна (права) грань

    % ------------------------------------------------------------
    % 3. ПЛАСТИНИ (нескінченно тонкі, напівпрозорі) -- малюємо ПІСЛЯ,
    %    щоб колір трохи змішався із сірим діелектриком там, де він є
    % ------------------------------------------------------------
    \draw[fill=red!50, fill opacity=0.7, draw=black]
        (0,0) -- (\Lp,0) -- ++(\ang:\W) -- ($(0,0)+(\ang:\W)$) -- cycle;    % верхня пластина (+Q)
    \draw[fill=blue!50, fill opacity=0.7, draw=black]
        (0,-\Z) -- (\Lp,-\Z) -- ++(\ang:\W) -- ($(0,-\Z)+(\ang:\W)$) -- cycle; % нижня пластина (-Q)

    % ------------------------------------------------------------
    % 4. ПІДПИСИ ЗАРЯДІВ ТА ДІЕЛЕКТРИКА
    % ------------------------------------------------------------
    \node at ({\Lp/2+\dxh/2}, {\dyh/2+0.05}) {$+Q$};
    \node[below=2pt] at ({0.32*\Lp}, -\Z) {$-Q$};
    \node at ({(\Dx+\Lp)/2}, {-\Z/2}) {$\varepsilon_r$};

    % ------------------------------------------------------------
    % 5. РОЗМІРНІ ЛІНІЇ
    % ------------------------------------------------------------
    % L -- довжина пластин (по задній кромці)
    \draw[gray,<->] (\dxh,{\dyh+0.35}) -- node[above]{$L$} (\Lp+\dxh,{\dyh+0.35});

    % W -- глибина (вздовж напрямку перспективи), зліва від пластини
    \draw[gray,<->] ($(-0.35,0)$) -- node[above,sloped]{$W$} ($(-0.35,0)+(\ang:\W)$);

    % Z -- відстань між пластинами
    \draw[gray,<->] (-0.15,0) -- node[left]{$Z$} (-0.15,-\Z);

    % x -- довжина діелектрика, що ще перебуває між пластинами
    \draw[red,decorate,decoration={brace,amplitude=6pt,mirror}]
        (\Dx,-\Z-0.45) -- node[below=6pt]{$x$} (\Lp,-\Z-0.45);

\end{tikzpicture}
